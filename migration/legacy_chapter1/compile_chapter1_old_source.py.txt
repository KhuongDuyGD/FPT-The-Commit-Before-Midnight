def clean_markup(text):
    text = text.replace("`", "").strip()
    if text.startswith("**") and text.endswith("**"):
        return text[2:-2]
    if text.startswith("*") and text.endswith("*"):
        return text[1:-1]
    return text

def read_sections():
    text = SOURCE.read_text(encoding="utf-8")
    text = re.split(r"^# FPT: The Commit Before Midnight\n", text, maxsplit=1, flags=re.M)[1]
    sections = {}
    key = None
    for line in text.splitlines():
        if line.startswith("# "):
            key = line[2:].split(" — ")[0].split(". ")[0]
            sections[key] = []
        elif key is not None:
            sections[key].append(line)
    result = {}
    for key, lines in sections.items():
        items = []
        i = 0
        while i < len(lines):
            line = lines[i].strip()
            i += 1
            if not line or line == "---":
                continue
            speaker = re.match(r"\*\*([^*]+):\*\*\s*(.*)", line)
            if speaker or line == "[NARRATOR:]":
                name, body = speaker.groups() if speaker else ("NARRATOR", "")
                parts = [body] if body else []
                while (not body and i < len(lines) and lines[i].strip()
                       and not lines[i].strip().startswith(("→", "[", "**"))):
                    parts.append(lines[i].rstrip())
                    i += 1
                items.append(("say", name, "\n".join(p.rstrip() for p in parts).replace("`", "").strip()))
            elif re.match(r"\d+\. \*\*", line):
                caption = re.match(r"\d+\. \*\*(.*?)\*\*", line).group(1)
                items.append(("choice", caption, ""))
                rest = line.split("**", 2)[-1]
                if "[" in rest:
                    items.append(("effect", rest, ""))
            elif line.startswith("[SYS]") or line.startswith("[Hệ thống]"):
                body = line.split("]", 1)[1].strip()
                if not body:
                    while i < len(lines) and lines[i].strip():
                        body += ("\n" if body else "") + lines[i].strip()
                        i += 1
                items.append(("sys", clean_markup(body), ""))
            elif line.startswith("["):
                tag, _, body = line.partition("]")
                items.append((tag[1:], body.strip(), ""))
            else:
                items.append(("note", line, ""))
        result[key] = items
    return result


def quote(text):
    return json.dumps(text.replace("[", "[[").replace("{", "{{").replace("%", "%%"), ensure_ascii=False)


class Compiler:
    def __init__(self, sections):
        self.sections = sections
        self.lines = ["# Generated by tools/compile_chapter1.py from the supplied Chapter 1.",
                      "# Dialogue and captions are immutable; presentation and systems live separately.", ""]
        self.used = set()
        self.manifest = []

    def code(self, line="", indent=1):
        self.lines.append("    " * indent + line if line else "")

    def label(self, name):
        self.code()
        self.code("label " + name + ":", 0)

    def emit(self, section, start=0, end=None, indent=1):
        items = self.sections[section]
        for index in range(start, len(items) if end is None else end):
            kind, value, text = items[index]
            cue = EXPRESSION_CUES.get(section, {}).get(index)
            if cue:
                self.code('$ set_player_expression("' + cue + '")', indent)
            if kind == "say":
                self.code(SPEAKERS[value] + " " + quote(text), indent)
                self.used.add((section, index))
                self.manifest.append({"section": section, "index": index, "speaker": value,
                                      "character": SPEAKERS[value], "text": text, "kind": "say"})
            elif kind == "sys":
                if re.fullmatch(r"\[ENERGY [+-]?\d+\]", value):
                    self.effect(value, indent, section, index)
                else:
                    self.code("system_msg " + quote(value.replace("**", "")), indent)
                    self.used.add((section, index))
                    self.manifest.append({"section": section, "index": index, "speaker": None,
                                          "character": "system_msg", "text": value.replace("**", ""), "kind": "sys"})
            elif kind == "PAUSE":
                self.code("pause 0.35", indent)
            elif kind in ("note", "effect", "FLAG"):
                self.effect(value, indent, section, index)
            elif kind == "SFX":
                self.code("# [SFX] " + value, indent)
                cue = SOUNDS.get((section, index))
                if cue:
                    self.code('$ play_scene_sound("' + cue + '")', indent)
                if (section, index) == ("9", 2):
                    self.code('$ scene_ambience()', indent)
            elif kind == "ACT":
                self.code("# [ACT] " + value, indent)
                for line in ACTIONS.get((section, index), []):
                    self.code(line, indent)
            elif kind == "BG":
                self.code("# [BG] " + value, indent)
                for line in BACKGROUNDS[(section, index)]:
                    self.code(line, indent)
            elif kind.startswith("FLASHBACK"):
                if kind == "FLASHBACK END":
                    self.code("hide screen device_panel", indent)
                    self.code("scene bedroom_morning with fade", indent)
                    self.code('$ scene_ambience("room")', indent)
                    self.code('$ set_stage("player")', indent)

    def effect(self, value, indent, section=None, index=None):
        for amount in re.findall(r"\[ENERGY ([+-]?\d+)\]", value):
            adjusted = ENERGY_OVERRIDES.get((section, index), int(amount))
            self.code("$ change_energy(" + ("+" if adjusted > 0 else "") + str(adjusted) + ")", indent)
        for name, amount in re.findall(r"\[AFFINITY: (LINH|NGÂN) ([+-]\d+)\]", value):
            self.code('$ change_trust("' + {"LINH": "linh", "NGÂN": "ngan"}[name] + '", ' + amount + ')', indent)
        flag = re.search(r"(?:FLAG:\s*|`)([A-Z_]+) (?:= (TRUE)|([+]\d+))", value)
        if flag:
            name, enabled, amount = flag.groups()
            if name not in {"OVERSLEPT", "RESPONSIBLE_ONCE", "CLASS_LATE", "COFFEE_ONLY",
                            "STILL_NO_REGRET", "FAKE_PROMISE", "DEADLINE_GREED", "TOO_MUCH_COFFEE",
                            "DEV_RESPECT", "PE_PRESSURE", "PE_CLUB_CAPTURED"}:
                raise ValueError("Unknown source flag " + name)
            self.code("$ " + name.lower() + (" = True" if enabled else " += " + amount), indent)

    def option(self, section, index, start=None, end=None, indent=1, extra=()):
        kind, caption, _ = self.sections[section][index]
        assert kind == "choice"
        self.code(quote(caption) + ":", indent + 1)
        self.used.add((section, index))
        self.manifest.append({"section": section, "index": index, "text": caption, "kind": "choice"})
        for line in extra:
            self.code(line, indent + 2)
        if start is not None:
            self.emit(section, start, end, indent + 2)

    def checkpoint(self):
        self.code("call ch1_campus_checkpoint")
        self.code("if _return:")
        self.code("jump ch1_infirmary", 2)

    def write(self):
        expected = {(section, i) for section, items in self.sections.items()
                    if section not in ("0", "16", "17", "18")
                    for i, item in enumerate(items) if item[0] in ("say", "choice", "sys")
                    and item[1] != "[ENERGY -10]"}
        assert self.used == expected, "Uncompiled source tokens: " + str(expected - self.used)
        for folder in (ROOT / "game/chapters", ROOT / "tests/rebuild"):
            folder.mkdir(parents=True, exist_ok=True)
        (ROOT / "game/chapters/chapter_01.rpy").write_text("\n".join(self.lines) + "\n", encoding="utf-8")
        data = {"source_sha256": hashlib.sha256(SOURCE.read_bytes()).hexdigest(), "content": self.manifest}
        (ROOT / "tests/rebuild/source_manifest.json").write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")
        print("Compiled", len(self.manifest), "exact source dialogue/system/choice entries.")


def scene(image, cast, ambience=None, expression=None):
    lines = ['scene ' + image, '$ scene_ambience(' + (quote(ambience) if ambience else "") + ')']
    if expression == "energy":
        lines.append('$ set_energy_expression()')
    elif expression:
        lines.append('$ set_player_expression("' + expression + '")')
    return lines + ['$ set_stage(' + cast + ')', 'with Dissolve(0.3)']
