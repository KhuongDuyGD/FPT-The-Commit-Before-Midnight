# Chapter navigation uses Ren'Py Start so direct selection gets a fresh save
# context. Continuing after Chapter 1 uses Jump within the current playthrough.
screen chapter_select():
    tag menu
    add "school_gate_afternoon"
    add Solid("#0b1327e8")
    text "CHAPTER SELECT" xalign 0.5 ypos 64 size 46 bold True
    vbox:
        xalign 0.5
        yalign 0.48
        spacing 24
        textbutton "Chương 1 — THE COMMIT BEFORE MIDNIGHT":
            action Start("start")
            xsize 1200
            padding (30, 22)
            text_size 29
            background Solid("#253659")
            hover_background Solid("#426195")
        textbutton ("Chương 2 — LOVE PROTOCOL" if chapter2_is_unlocked() else "Chương 2 — LOCKED"):
            action Start("chapter2_start")
            sensitive chapter2_is_unlocked()
            xsize 1200
            padding (30, 22)
            text_size 29
            text_insensitive_color "#929aaa"
            background Solid("#253659")
            hover_background Solid("#426195")
        if not chapter2_is_unlocked():
            text "Hoàn thành Good Ending Chương 1 để mở Chương 2." size 23 color "#bac7dc" xalign 0.5
    textbutton "Return" action Return() xalign 0.5 yalign 0.90 text_size 25

        xalign 0.5
        ypos CHOICE_BOTTOM
        yanchor 1.0
        xsize 980

        background Solid("#162944")

            spacing 22
            text "NGÂN · TIN NHẮN" size 25 color "#94dcff"
            text message size 28 xmaximum 916

            textbutton "Đã hiểu" action Hide("ngan_hint_message") text_size 25 xalign 1.0

screen chapter2_cg(kind, title, subtitle=""):
    # Keep the supplied CG visible. Unlike Chapter 1's opaque card, this panel
    # leaves the characters and scene readable above its bottom strip.
    modal True
    frame:
        yalign 1.0
        xfill True
        padding (30, 18)
        background Solid("#081020ed")
        vbox:
            xalign 0.5
            spacing 7
            text kind size 26 bold True xalign 0.5 color "#ffce83"
            text title size 37 xalign 0.5
            if subtitle:
                text subtitle size 20 text_align 0.5 xalign 0.5
            textbutton "Continue" action Return() xalign 0.5 text_size 24

screen chapter2_end_menu():
    modal True
    add Solid("#080d18ec")
    vbox:
        xalign 0.5
        yalign 0.5
        spacing 22
        text ("ROUTE COMPLETE" if game_route_complete else "CHƯƠNG 2 — THE END") xalign 0.5 size 42 bold True
        textbutton "Trở về Menu Chính" action MainMenu(confirm=False) xalign 0.5 text_size 28
        textbutton "Chapter Select" action ShowMenu("chapter_select") xalign 0.5 text_size 28
        textbutton "Ending Gallery" action ShowMenu("ending_gallery") xalign 0.5 text_size 25

screen chapter_gallery_contents():
    vbox:
        xpos 150
        ypos 160
        spacing 12
        for title, ending_id, unlocked in [
            ("HOME SAFE", "HOME_SAFE", "HOME_SAFE" in persistent.chapter1_routes),
            ("BLACK ALLOY CLUB", "BLACK_ALLOY_CLUB", "BLACK_ALLOY_CLUB" in persistent.chapter1_routes),
            ("WALKING CORPSE", "WALKING_CORPSE", "WALKING_CORPSE" in persistent.chapter1_routes),
            ("INFIRMARY", "INFIRMARY", "INFIRMARY" in persistent.chapter1_routes),
            ("LOVE PROTOCOL ESTABLISHED", "nghi_good", persistent.ending_nghi_good_unlocked),
            ("404 — LOVE NOT FOUND", "nghi_bad", persistent.ending_nghi_bad_unlocked),
            ("YOU WERE NEVER LOST", "ngan", persistent.ending_ngan_unlocked)]:
            frame:
                xsize 1620
                ysize 88
                padding (26, 16)
                background Solid("#19283d")
                hbox:
                    yalign 0.5
                    spacing 22
                    fixed:
                        xsize 1200
                        ysize 50
                        text title size 28 xmaximum 1200 yalign 0.5
                    if unlocked:
                        textbutton "View ending" action Show("gallery_detail", ending=ending_id) yalign 0.5 text_size 24
                    else:
                        text "LOCKED" color "#718299" size 24 yalign 0.5

screen ending_gallery():
    tag menu
    use menu_shell("ENDING GALLERY")
    use chapter_gallery_contents

screen gallery_detail(ending):
    modal True
    zorder 120
    add Solid("#080e19ed")
    vbox:
        xalign 0.5
        yalign 0.5
        spacing 25
        if ending in ROUTE_PRIORITY:
            text ending.replace("_", " ") size 52 xalign 0.5 color "#f3be85"
        else:
            add ("cg_nghi_good" if ending == "nghi_good" else "cg_ngan" if ending == "ngan" else "nearby_pub_night") xysize (1280, 720)
        textbutton "Close" action Hide("gallery_detail") xalign 0.5
